从 MySQL 8.0 开始，默认的身份验证插件是 caching_sha2_password，它在某些情况下（如使用 SSL 或新用户）会要求客户端支持 公钥检索（public key retrieval） 来完成密码交换。

添加 allowPublicKeyRetrieval=true 参数以允许公钥检索，并确保客户端支持 SSL。

$ cat sso.conf
#HTTPS使用SSL访问的配置
server {
    #SSL 默认访问端口号为 443
    listen 443 ssl;

    #请填写绑定证书的域名
    server_name sso.abc.com;

    #请填写证书文件的相对路径或绝对路径
    ssl_certificate /etc/letsencrypt/live/sso.abc.com/fullchain.pem;

    #请填写私钥文件的相对路径或绝对路径
    ssl_certificate_key /etc/letsencrypt/live/sso.abc.com/privkey.pem;

    #缓存SSL握手产生的参数和加密密钥的时长
    ssl_session_timeout 5m;

    #请按照以下协议配置
    ssl_protocols TLSv1.2 TLSv1.3;

    #请按照以下套件配置，配置加密套件，写法遵循 openssl 标准。
    #ssl_ciphers ECDHE-RSA-AES128-GCM-SHA256:HIGH:!aNULL:!MD5:!RC4:!DHE;
    ssl_ciphers ECDHE-RSA-AES128-GCM-SHA256:ECDHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES128-GCM-SHA256;

    #启用 SSL 会话缓存，避免每次 HTTPS 请求都进行完整的 TLS 握手（耗时且消耗 CPU）
    ssl_session_cache shared:SSL:10m;

    #加密套件优先选择服务器的加密套件
    ssl_prefer_server_ciphers off;

    location / {
        proxy_pass http://127.0.0.1:8081;  # 转发到 Keycloak
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Port $server_port;

        # WebSocket 支持（Keycloak Admin Console 需要）
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";

        # 超时设置（避免 Keycloak 登录卡住）
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
}

cat frpc.toml
# frpc.toml - frp 客户端配置

# 服务端连接配置
serverAddr = "1.23.45.67"              # ⚠️ 改为你的服务器 IP
serverPort = 7000

# 认证配置（必须与服务端一致）
auth.method = "token"
auth.token = "token密码"    # ⚠️ 必须与服务端一致

# 日志配置
log.to = "./frpc.log"
log.level = "info"
log.maxDays = 3

# Keycloak HTTPS 代理
# [[proxies]]
# name = "keycloak-https"
# type = "https"
# localIP = "127.0.0.1"
# localPort = 8081
# customDomains = ["frp.abc.com"]     # ⚠️ 改为你的域名

# 如果需要同时支持 HTTP（不推荐）
[[proxies]]
name = "keycloak-http"
type = "http"
localIP = "127.0.0.1"
localPort = 8081
customDomains = ["frp.abc.com"]

# 👇 添加这个部分，强制设置 HTTPS 相关的请求头
[proxies.requestHeaders]
[proxies.requestHeaders.set]
X-Forwarded-Proto = "https"
X-Forwarded-Host = "frp.abc.com"
X-Forwarded-Port = "443"

添加了强制设置 HTTPS 相关的请求头，问题解决！ 访问https://frp.abc.com/realms/master/.well-known/openid-configuration提示"issuer":"https://frp.abc.com/realms/master"。

    ssl_certificate_key /etc/letsencrypt/live/sso.abc.com/privkey.pem;

    #缓存SSL握手产生的参数和加密密钥的时长
    ssl_session_timeout 5m;

    #请按照以下协议配置
    ssl_protocols TLSv1.2 TLSv1.3;

    #请按照以下套件配置，配置加密套件，写法遵循 openssl 标准。
    #ssl_ciphers ECDHE-RSA-AES128-GCM-SHA256:HIGH:!aNULL:!MD5:!RC4:!DHE;
    ssl_ciphers ECDHE-RSA-AES128-GCM-SHA256:ECDHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES128-GCM-SHA256;

    #启用 SSL 会话缓存，避免每次 HTTPS 请求都进行完整的 TLS 握手（耗时且消耗 CPU）
